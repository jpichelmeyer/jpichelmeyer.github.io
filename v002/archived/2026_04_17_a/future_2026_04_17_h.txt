    function loadPOSS(fileContent) {
        const lines = fileContent.split('\n');
        let state = {};

        lines.forEach(line => {
            const [key, value] = line.split(':').map(s => s.trim());
            if (key && value) {
                state[key] = value;
                // Apply to your OS, e.g., if (key === 'bg') document.body.style.background = value;
            }
        });
    }
